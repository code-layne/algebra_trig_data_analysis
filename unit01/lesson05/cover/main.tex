\begin{learningtargetbox}
\small
\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item \ldots{} recognize and factor a \textbf{difference of squares}, $a^2-b^2 = (a+b)(a-b)$, and
        explain why a \textbf{sum} of squares is prime over the integers.
  \item \ldots{} recognize and factor a \textbf{perfect square trinomial},
        $a^2 \pm 2ab + b^2 = (a \pm b)^2$, by testing the middle term.
  \item \ldots{} factor a \textbf{sum or difference of cubes},
        $a^3 \pm b^3 = (a \pm b)(a^2 \mp ab + b^2)$.
  \item \ldots{} factor \textbf{completely}, GCF first, and \textbf{verify a polynomial identity} by
        multiplying the factored form back out.
\end{enumerate}
\end{learningtargetbox}
