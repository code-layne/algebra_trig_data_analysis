\begin{notesbox}{Exponent Rules and Polynomial Operations --- On Your Own}
\small
Notes closed. Three items.

\begin{enumerate}[leftmargin=1.6em, itemsep=8pt, topsep=6pt]

\item Simplify, writing your answer with no negative exponents: $\dfrac{12a^5b^3}{3a^2b^7}$
\begin{work}
  \dfrac{12a^5b^3}{3a^2b^7} &= 4\,a^{5-2}\,b^{3-7} \\
                            &= 4a^3b^{-4} \\
                            &= \dfrac{4a^3}{b^4}
\end{work}

\item Subtract and write in standard form: $\left(5x^2-3x+7\right)-\left(2x^2+4x-9\right)$
\begin{work}
  \left(5x^2-3x+7\right)-\left(2x^2+4x-9\right) &= 5x^2-3x+7-2x^2-4x+9 \\
                                                &= 3x^2-7x+16
\end{work}

\item A classmate writes $\left(3x^2\right)^3 = 3x^6$. Give the correct answer, then explain in
one sentence what the classmate forgot.
\begin{work}
  \left(3x^2\right)^3 &= 3^3\left(x^2\right)^3 \\
                      &= 27x^6
\end{work}
\ansline{The outer exponent applies to the coefficient $3$ as well, giving $27$.}

\end{enumerate}
\end{notesbox}
